```
#!/bin/bash
# Kimi-K3 MIXED deployment (no PD split): 4 nodes, tp=16 pp=4, DeepEP ON, DSPARK OFF.
# Purpose: validate sglang PR#34735 + sgl-kernel-npu PR#715 (pipeline parallel for K3).
#   PR-on  sglang:  /home/<USER>/personal/sglang        (branch k3-pp)
#   PR-on  kernel:  sgl_kernel_npu installed in container  (>= 2026-08-12 build, has #715)
#   PR-off: PYTHONPATH_SGLANG -> base sglang checkout; reinstall base sgl_kernel_npu (todo, code not pulled yet)
#
# Topology: rank0 @ <NODE_A>  rank1 @ <NODE_B>  rank2 @ <NODE_C>  rank3 @ <NODE_D>
#   93 layers -> SGLANG_PP_LAYER_PARTITION=24,23,23,23 ; TP16 x PP4 x DP1 = 64 devices.
#
# Run inside container <CONTAINER> on each node:
#   NODE_RANK=0 bash mixed.sh   # <NODE_A>
#   NODE_RANK=1 bash mixed.sh   # <NODE_B>
#   NODE_RANK=2 bash mixed.sh   # <NODE_C>
#   NODE_RANK=3 bash mixed.sh   # <NODE_D>
#
# Lesson from round8 COMPARE notes: chunk=2048 + PP hangs (DeepEP on AND off).
# Use CHUNKED_PREFILL_SIZE=4096. Beware AtbRingMLA OOM under conc; HCCL leftovers -> clean kills.

set -o pipefail

SCRIPT_DIR="$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd)"
LOG_DIR="${LOG_DIR:-${SCRIPT_DIR}/logs}"
PID_DIR="${PID_DIR:-${SCRIPT_DIR}/pids}"
mkdir -p "$LOG_DIR" "$PID_DIR"

ACTION="${1:-start}"
MODEL_PATH="${MODEL_PATH:-<MODEL_PATH>"
# PR switch: point PYTHONPATH_SGLANG at the base checkout for the PR-off leg.
#   PR-on  default = personal (branch k3-pp, includes #34735)
#   PR-off          = community (base; user pulls code there), set PYTHONPATH_SGLANG
PYTHONPATH_SGLANG="${PYTHONPATH_SGLANG:-/home/<USER>/community/sglang/python}"

RANK_IPS=("<NODE_A>" "<NODE_B>" "<NODE_C>" "<NODE_D>")
NNODES="${NNODES:-4}"
RANK0_IP="${RANK_IPS[0]}"
# MIXED_PORT distinct from profile (31001/31002) and 30000/5001/5002/8000.
MIXED_PORT="${MIXED_PORT:-31003}"
DIST_INIT="${DIST_INIT:-${RANK0_IP}:15003}"

USE_DEEPEP="${USE_DEEPEP:-1}"
TP_SIZE="${TP_SIZE:-16}"
PP_SIZE="${PP_SIZE:-4}"
MEM_FRACTION="${MEM_FRACTION:-0.85}"
CHUNKED_PREFILL_SIZE="${CHUNKED_PREFILL_SIZE:-4096}"
MAX_RUNNING="${MAX_RUNNING:-4}"
DISABLE_CUDA_GRAPH="${DISABLE_CUDA_GRAPH:-1}"
HCCL_BUFFSIZE="${HCCL_BUFFSIZE:-800}"
# CUDA graph bucket bs; EVERY value must be a mult·iple of attn_tp (16) or it gets
# filtered out (base_cuda_graph_runner `bs % alignment == 0`). Start small on 93L/pp4.
CUDA_GRAPH_BS="${CUDA_GRAPH_BS:-16 32}"

detect_my_ip() {
  hostname -I 2>/dev/null | tr ' ' '\n' | grep -E '^<SUBNET>.(<NODE_A>|<NODE_B>|<NODE_C>|<NODE_D>)$' | head -1
}

MY_IP="$(detect_my_ip)"
if [[ -z "${NODE_RANK:-}" ]]; then
  for i in "${!RANK_IPS[@]}"; do
    [[ "$MY_IP" == "${RANK_IPS[$i]}" ]] && NODE_RANK="$i" && break
  done
fi
if [[ -z "${NODE_RANK}" || "$NODE_RANK" -lt 0 || "$NODE_RANK" -ge "$NNODES" ]]; then
  echo "ERROR: set NODE_RANK=0..$((NNODES-1)) or run on ${RANK_IPS[*]} (got NODE_RANK=${NODE_RANK:-empty})"
  exit 1
fi
EXPECTED_IP="${RANK_IPS[$NODE_RANK]}"
IFACE_IP="${MY_IP:-$EXPECTED_IP}"

stop_all() {
  if [[ -f "${PID_DIR}/mixed.pid" ]]; then
    pid="$(cat "${PID_DIR}/mixed.pid")"
    kill "$pid" 2>/dev/null || true
    sleep 2
    kill -9 "$pid" 2>/dev/null || true
    rm -f "${PID_DIR}/mixed.pid"
  fi
  echo "mixed stopped"
}

if [[ "$ACTION" == "stop" ]]; then
  stop_all
  exit 0
fi

echo performance | tee /sys/devices/system/cpu/cpu*/cpufreq/scaling_governor >/dev/null || true
sysctl -w vm.swappiness=10 || true
sysctl -w kernel.numa_balancing=0 || true
sysctl -w kernel.sched_migration_cost_ns=50000 || true

export SGLANG_SET_CPU_AFFINITY=1
export SGLANG_ONE_VISIBLE_DEVICE_PER_PROCESS=1
export SGLANG_NPU_USE_TRITON_PREFIX_KV_CACHE_STORE=1
export SGLANG_NPU_USE_MULTI_STREAM=0

unset https_proxy http_proxy HTTPS_PROXY HTTP_PROXY ASCEND_LAUNCH_BLOCKING
unset SGLANG_ENABLE_SPEC_V2 SGLANG_RAGGED_VERIFY_MODE || true
# DSPARK conflicts with PP: never enable speculative here.
unset SGLANG_PP_LAYER_PARTITION || true

source /usr/local/Ascend/ascend-toolkit/set_env.sh
source /usr/local/Ascend/nnal/atb/set_env.sh

export PYTORCH_NPU_ALLOC_CONF="${PYTORCH_NPU_ALLOC_CONF:-expandable_segments:True}"
export STREAMS_PER_DEVICE=32
# From prefill working recipe; DeepEP-normal needs a small-ish HCCL, raise if tiling says too SMALL.
export HCCL_BUFFSIZE="${HCCL_BUFFSIZE}"
export HCCL_OP_EXPANSION_MODE=AIV
export PYTHONPATH="${PYTHONPATH_SGLANG}:${PYTHONPATH:-}"
export SGLANG_NPU_PROFILING=0
export SGLANG_ENABLE_OVERLAP_PLAN_STREAM=0
# 24,23,23,23 = 93 full layers across PP4 (full model; not the 24-layer test slice).
export SGLANG_PP_LAYER_PARTITION="${SGLANG_PP_LAYER_PARTITION:-24,23,23,23}"

MOE_A2A_ARGS=()
GRAPH_ARGS=()
if [[ "$USE_DEEPEP" == "1" ]]; then
  export DEEP_NORMAL_MODE_USE_INT8_QUANT=1
  # Kernel PR#715 enables multi-round dispatch (>512 experts). K3 has 896 -> triggers it.
  export SGLANG_DEEPEP_NUM_MAX_DISPATCH_TOKENS_PER_RANK="${SGLANG_DEEPEP_NUM_MAX_DISPATCH_TOKENS_PER_RANK:-64}"
  export DEEPEP_NORMAL_LONG_SEQ_ROUND="${DEEPEP_NORMAL_LONG_SEQ_ROUND:-64}"
  export DEEPEP_NORMAL_LONG_SEQ_PER_ROUND_TOKENS="${DEEPEP_NORMAL_LONG_SEQ_PER_ROUND_TOKENS:-512}"
  MOE_A2A_ARGS=(--moe-a2a-backend deepep --deepep-mode auto)
fi
# DeepEP or not: on this K3 MoE the decode graph bucket must be a multiple of
# attn_tp(16) (TP-gathered buffer), otherwise `bs % alignment == 0` filters it out.
# shellcheck disable=SC2206
GRAPH_ARGS=(--cuda-graph-bs ${CUDA_GRAPH_BS[@]})
if [[ "${DISABLE_CUDA_GRAPH}" == "1" ]]; then
  GRAPH_ARGS+=(--disable-cuda-graph)
fi

find_iface_by_ip() {
  python3 - "$1" <<'PY'
import fcntl, os, socket, struct, sys
target = sys.argv[1]
def iface_ip(name):
    s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    try:
        return socket.inet_ntoa(fcntl.ioctl(s.fileno(), 0x8915, struct.pack("256s", name[:15].encode()))[20:24])
    except OSError:
        return None
    finally:
        s.close()
for name in sorted(os.listdir("/sys/class/net")):
    if name != "lo" and iface_ip(name) == target:
        print(name); break
PY
}

IFACE="$(find_iface_by_ip "${IFACE_IP}")"
[[ -n "$IFACE" ]] || { echo "ERROR: no NIC for ${IFACE_IP}"; exit 1; }
export HCCL_SOCKET_IFNAME="${IFACE}"
export GLOO_SOCKET_IFNAME="${IFACE}"

TS="$(date +%Y-%m-%d_%H-%M-%S)"
MIX_LOG="${LOG_DIR}/mixed_rank${NODE_RANK}_${TS}.log"
echo "Mixed rank=${NODE_RANK} ip=${IFACE_IP} iface=${IFACE} deepep=${USE_DEEPEP} (DSPARK=off)"
echo "  model=${MODEL_PATH} tp=${TP_SIZE} pp=${PP_SIZE} nnodes=${NNODES}"
echo "  partition=${SGLANG_PP_LAYER_PARTITION} mem=${MEM_FRACTION} chunk=${CHUNKED_PREFILL_SIZE}"
echo "  dist=${DIST_INIT} http=:${MIXED_PORT}"
echo "  PYTHONPATH_SGLANG=${PYTHONPATH_SGLANG} log=${MIX_LOG}"

sglang serve \
  --model-loader-extra-config '{"enable_multithread_load": true}' \
  --dist-init-addr "${DIST_INIT}" \
  --nnodes "${NNODES}" \
  --node-rank "${NODE_RANK}" \
  --model-path "$MODEL_PATH" \
  --tokenizer-path "$MODEL_PATH" \
  --trust-remote-code \
  --attention-backend ascend \
  --device npu \
  --quantization modelslim \
  --dtype bfloat16 \
  --tp-size "${TP_SIZE}" \
  --pp-size "${PP_SIZE}" \
  --dp-size 1 \
  --base-gpu-id 0 \
  --disable-radix-cache \
  --mem-fraction-static "${MEM_FRACTION}" \
  --chunked-prefill-size "${CHUNKED_PREFILL_SIZE}" \
  --max-running-requests "${MAX_RUNNING}" \
  --host <LISTEN_ADDR> \
  --port "${MIXED_PORT}" \
  --reasoning-parser kimi_k3 \
  "${GRAPH_ARGS[@]}" \
  "${MOE_A2A_ARGS[@]}" \
  --watchdog-timeout 9000 \
  2>&1 | tee "$MIX_LOG"
```
